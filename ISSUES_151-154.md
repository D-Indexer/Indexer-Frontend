## Issue #151: Generate Typed Contract Bindings from Spec

**Type:** Technical Debt / Type Safety  
**Priority:** High (Foundation)  
**Component:** `src/services/stellar/stellarService.ts`

### Summary
Replace manual `nativeToScVal` / `scValToNative` conversions with typed contract bindings generated from the deployed Folder contract spec using `stellar contract bindings typescript`, ensuring argument types and return shapes are checked at compile time.

### Current Behavior / Motivation

**Manual Encoding** (`src/services/stellar/stellarService.ts`):

```typescript
function toArgs(method: string, args: any[]): xdr.ScVal[] {
  if (method === 'claim_folder') {
    // Hand-encode every argument
    return [
      nativeToScVal(args[0], { type: 'address' }),  // owner
      nativeToScVal(args[1], { type: 'string' }),   // ipfs_cid
      nativeToScVal(args[2], { type: 'u32' }),      // template_id
    ];
  }
  // ... more methods
}

async function getOnChainPortfolio(owner: string): Promise<Portfolio> {
  const result = await invoke('get_portfolio', [owner]);
  
  // Cast without type safety!
  const portfolio = scValToNative(result) as {
    ipfs_cid: string;
    template_id: number;
    minted_at: number;
  };
  
  return portfolio;
}
```

**Problems:**
- Contract signature changes (e.g., `template_id: u32` → `template_id: u64`) compile fine but fail at runtime
- `as` casts bypass TypeScript safety
- Manual encoding error-prone
- No compile-time validation

## Issue #152: Show Simulated Fees Before Signing

**Type:** UX Enhancement / Fee Transparency  
**Priority:** High  
**Component:** `src/services/stellar/stellarService.ts`, `src/features/transactions/components/TxStatus.tsx`

### Summary
Expose the Soroban resource fee from simulation and the network inclusion fee from `getFeeStats` to the user **before** Freighter opens, and automatically raise the inclusion fee when the network is in surge pricing.

### Current Behavior / Motivation

**Current Flow:**

```typescript
// File: src/services/stellar/stellarService.ts

export async function invoke(method: string, args: xdr.ScVal[]): Promise<string> {
  // 1. Build with BASE_FEE (100 stroops)
  const tx = new TransactionBuilder(account, {
    fee: BASE_FEE,  // ← User doesn't see this
    networkPassphrase: Networks.TESTNET
  });
  
  // 2. Simulate
  const simulation = await rpc.simulateTransaction(tx);
  
  // 3. Assemble (adds resource fee silently)
  const assembled = SorobanRpc.assembleTransaction(tx, simulation);
  //                ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
  //                Resource fee added here, user never sees it
  
  // 4. Sign (Freighter shows TOTAL fee for first time)
  const signed = await signTransaction(assembled.toXDR());
  
  // 5. Send
  return await sendTransaction(signed);
}
```

**Problems:**
- User sees final fee **only inside Freighter**
- No opportunity to cancel before signing if fee is too high
- **Surge pricing:** When network is congested, `BASE_FEE` causes `txINSUFFICIENT_FEE` or `TRY_AGAIN_LATER`
- Current error handling maps this to generic "network is busy" message
- No fee bumping strategy

### Proposed Implementation

#### Step 1: Add Fee Estimation Before Signing

```typescript
// File: src/services/stellar/stellarService.ts

interface FeeEstimate {
  inclusionFee: string;      // Base fee per operation (from getFeeStats)
  resourceFee: string;        // Resource fee from simulation
  totalFee: string;           // Sum in stroops
  totalFeeXLM: string;        // Human-readable XLM
  isSurge: boolean;           // Network in surge pricing
  recommendedInclusionFee: string;  // Bumped fee if surge
}

export async function estimateFees(
  method: string,
  args: xdr.ScVal[]
): Promise<FeeEstimate> {
  // 1. Get current network fee stats
  const feeStats = await rpc.getFeeStats();
  
  // 2. Determine if network is in surge pricing
  const p50 = Number(feeStats.sorobanInclusionFee.p50);
  const p95 = Number(feeStats.sorobanInclusionFee.p95);
  const isSurge = p95 > p50 * 1.5;  // 50% higher = surge
  
  // 3. Use p95 during surge, p50 otherwise
  const inclusionFee = isSurge ? p95 : p50;
  
  // 4. Build transaction with recommended fee
  const account = await loadAccount(await getPublicKey());
  const tx = new TransactionBuilder(account, {
    fee: inclusionFee.toString(),
    networkPassphrase: Networks.TESTNET
  })
    .addOperation(/* soroban operation */)
    .setTimeout(300)
    .build();
  
  // 5. Simulate to get resource fee
  const simulation = await rpc.simulateTransaction(tx);
  
  if (!SorobanRpc.Api.isSimulationSuccess(simulation)) {
    throw new Error('Simulation failed');
  }
  
  const resourceFee = simulation.minResourceFee;
  const totalFee = BigInt(inclusionFee) + BigInt(resourceFee);
  
  return {
    inclusionFee: inclusionFee.toString(),
    resourceFee: resourceFee,
    totalFee: totalFee.toString(),
    totalFeeXLM: (Number(totalFee) / 10_000_000).toFixed(7),  // stroops → XLM
    isSurge,
    recommendedInclusionFee: inclusionFee.toString()
  };
}

export async function invoke(
  method: string,
  args: xdr.ScVal[],
  feeEstimate?: FeeEstimate  // ← Pass pre-computed estimate
): Promise<string> {
  // If no estimate provided, compute one
  if (!feeEstimate) {
    feeEstimate = await estimateFees(method, args);
  }
  
  // Build with recommended fee
  const tx = new TransactionBuilder(account, {
    fee: feeEstimate.recommendedInclusionFee,
    networkPassphrase: Networks.TESTNET
  });
  
  // ... rest of invoke logic
}
```

#### Step 4: Handle Insufficient Fee Errors Gracefully

```typescript
// File: src/services/stellar/stellarService.ts

export async function sendTransaction(signedXDR: string): Promise<string> {
  try {
    const response = await rpc.sendTransaction(signedXDR);
    
    if (response.status === 'PENDING') {
      return response.hash;
    }
    
    throw new Error(response.status);
    
  } catch (error) {
    // Detect insufficient fee errors
    if (error.message.includes('txINSUFFICIENT_FEE') ||
        error.message.includes('TRY_AGAIN_LATER')) {
      
      throw new TransactionError(
        'Transaction fee was too low for current network conditions. ' +
        'The network may be congested. Please try again.',
        {
          code: 'INSUFFICIENT_FEE',
          recoverable: true
        }
      );
    }
    
    throw error;
  }
}
```

## Issue #153: Automatically Restore Archived Contract State

**Type:** Enhancement / Auto-Recovery  
**Priority:** High  
**Component:** `src/services/stellar/stellarService.ts`

### Summary
When simulation returns a restore preamble (`isSimulationRestore = true`), automatically build and submit a `RestoreFootprint` transaction, then re-simulate and continue the original call, rather than telling the user to "try again later".

### Current Behavior / Motivation

**Current Code:**

```typescript
// File: src/services/stellar/stellarService.ts

export async function invoke(method: string, args: xdr.ScVal[]): Promise<string> {
  const simulation = await rpc.simulateTransaction(tx);
  
  // ❌ Throws instead of restoring
  if (SorobanRpc.Api.isSimulationRestore(simulation)) {
    throw new TransactionError(
      'Some contract state has expired. Please try again later.',
      { code: 'ARCHIVED_STATE' }
    );
  }
  
  // ... continue
}
```

**Problems:**
- User returning after months of inactivity is **permanently blocked**
- "Try again later" doesn't help — nobody restores the entries
- No automatic recovery path
- Poor UX for dormant accounts

### Proposed Implementation

#### Step 1: Detect and Handle Restore Simulation

```typescript
// File: src/services/stellar/stellarService.ts

export type TxStage = 
  | 'idle'
  | 'estimating'
  | 'restoring'  // ← New stage
  | 'signing'
  | 'confirming'
  | 'success'
  | 'error';

export async function invoke(
  method: string,
  args: xdr.ScVal[]
): Promise<string> {
  // 1. Initial simulation
  let simulation = await rpc.simulateTransaction(tx);
  
  // 2. Handle restore preamble
  if (SorobanRpc.Api.isSimulationRestore(simulation)) {
    console.log('Archived state detected, restoring...');
    
    // Notify UI
    emitStageChange('restoring');
    
    // 3. Build and submit RestoreFootprint tx
    await restoreArchivedState(simulation.restorePreamble);
    
    // 4. Reload account (sequence number changed)
    account = await loadAccount(await getPublicKey());
    
    // 5. Re-simulate original transaction
    const newTx = new TransactionBuilder(account, {
      fee: simulation.minResourceFee,
      networkPassphrase: Networks.TESTNET
    })
      .addOperation(/* same operation */)
      .build();
    
    simulation = await rpc.simulateTransaction(newTx);
    
    if (!SorobanRpc.Api.isSimulationSuccess(simulation)) {
      throw new Error('Simulation failed after restore');
    }
  }
  
  // 6. Continue with original flow
  const assembled = SorobanRpc.assembleTransaction(tx, simulation);
  
  emitStageChange('signing');
  const signed = await signTransaction(assembled.toXDR());
  
  emitStageChange('confirming');
  return await sendTransaction(signed);
}

async function restoreArchivedState(
  preamble: SorobanRpc.Api.RestorePreamble
): Promise<void> {
  const account = await loadAccount(await getPublicKey());
  
  // 1. Build RestoreFootprint transaction
  const restoreTx = new TransactionBuilder(account, {
    fee: preamble.minResourceFee,
    networkPassphrase: Networks.TESTNET
  })
    .addOperation(
      Operation.restoreFootprint({})  // Empty operation
    )
    .setSorobanData(preamble.transactionData.build())  // ← Key: use preamble data
    .setTimeout(300)
    .build();
  
  // 2. Sign restore transaction
  const signedXDR = await signTransaction(restoreTx.toXDR());
  
  // 3. Submit and poll
  const response = await rpc.sendTransaction(signedXDR);
  
  if (response.status !== 'PENDING') {
    throw new Error(`Restore failed: ${response.status}`);
  }
  
  // 4. Wait for confirmation
  await pollTransaction(response.hash);
  
  console.log('State restored successfully');
}
```

#### Step 1: Create Pending Transaction Store

```typescript
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

export interface PendingTx {
  hash: string;
  method: string;
  owner: string;  // Address that submitted
  submittedAt: number;  // Unix timestamp
  timebound: number;  // Transaction timebound (submittedAt + 300s)
}

interface PendingTxStore {
  pending: Record<string, PendingTx>;  // Keyed by hash
  
  add: (tx: PendingTx) => void;
  remove: (hash: string) => void;
  getByOwnerAndMethod: (owner: string, method: string) => PendingTx | undefined;
  pruneStale: () => void;
}

export const usePendingTxStore = create<PendingTxStore>()(
  persist(
    (set, get) => ({
      pending: {},
      
      add: (tx) => set((state) => ({
        pending: {
          ...state.pending,
          [tx.hash]: tx
        }
      })),
      
      remove: (hash) => set((state) => {
        const { [hash]: removed, ...remaining } = state.pending;
        return { pending: remaining };
      }),
      
      getByOwnerAndMethod: (owner, method) => {
        const pending = get().pending;
        return Object.values(pending).find(
          (tx) => tx.owner === owner && tx.method === method
        );
      },
      
      pruneStale: () => set((state) => {
        const now = Date.now();
        const fresh = Object.entries(state.pending).filter(
          ([_, tx]) => tx.timebound + 60000 > now  // Keep for 1 min after timebound
        );
        return { pending: Object.fromEntries(fresh) };
      })
    }),
    {
      name: 'pending-transactions',  // localStorage key
      version: 1
    }
  )
);
```

#### Step 3: Resume Polling on App Start

**File:** `src/App.tsx` or `src/features/transactions/components/PendingTxWatcher.tsx`

```typescript
import { useEffect } from 'react';
import { usePendingTxStore } from '@/store/pendingTxStore';
import { rpc } from '@/services/stellar/stellarService';

export function PendingTxWatcher() {
  const { pending, remove, pruneStale } = usePendingTxStore();
  const [resolvingHashes, setResolvingHashes] = useState<Set<string>>(new Set());
  
  useEffect(() => {
    // Prune stale entries on mount
    pruneStale();
    
    // Resume polling for each pending tx
    Object.values(pending).forEach((tx) => {
      if (!resolvingHashes.has(tx.hash)) {
        setResolvingHashes((prev) => new Set(prev).add(tx.hash));
        resumePolling(tx);
      }
    });
  }, []);
  
  async function resumePolling(tx: PendingTx) {
    try {
      // Poll with exponential backoff
      for (let i = 0; i < 30; i++) {
        const result = await rpc.getTransaction(tx.hash);
        
        if (result.status === 'SUCCESS') {
          console.log(`Transaction ${tx.hash} succeeded`);
          remove(tx.hash);
          
          // Invalidate relevant queries (React Query, SWR, etc.)
          queryClient.invalidateQueries(['portfolio', tx.owner]);
          
          return;
        }
        
        if (result.status === 'FAILED') {
          console.error(`Transaction ${tx.hash} failed`);
          remove(tx.hash);
          return;
        }
        
        // Exponential backoff: 1s, 2s, 4s, 8s, ...
        const delay = Math.min(1000 * Math.pow(2, i), 30000);
        await sleep(delay);
      }
      
      // Timeout: keep in store with pending: true
      console.warn(`Transaction ${tx.hash} still pending after 30 attempts`);
      
    } catch (error) {
      console.error(`Error polling transaction ${tx.hash}:`, error);
    } finally {
      setResolvingHashes((prev) => {
        const next = new Set(prev);
        next.delete(tx.hash);
        return next;
      });
    }
  }
  
  // Show banner if any tx is still resolving
  const activeCount = resolvingHashes.size;
  
  if (activeCount === 0) return null;
  
  return (
    <div className="fixed top-0 left-0 right-0 bg-blue-600 text-white px-4 py-2 text-center z-50">
      <div className="flex items-center justify-center gap-2">
        <Spinner size="sm" />
        <span>
          {activeCount} transaction{activeCount > 1 ? 's' : ''} confirming...
        </span>
        <a
          href={`https://stellar.expert/explorer/testnet/tx/${Object.keys(pending)[0]}`}
          target="_blank"
          rel="noopener noreferrer"
          className="underline ml-2"
        >
          View on Explorer
        </a>
      </div>
    </div>
  );
}
```

Pending transactions survive page reloads. If you close the browser during confirmation:
