## Issue #155: Timebound-Aware Exponential Backoff Polling

### Problem Statement

**Current Behavior:**
```typescript
// Fixed 30 attempts, no relation to transaction timebound
pollTransaction(sent.hash, { 
  attempts: 30, 
  sleepStrategy: rpc.BasicSleepStrategy 
});

// Problems:
// 1. Transaction has 120s timebound (maxTime)
// 2. May give up before timebound expires (premature failure)
// 3. May poll after timebound expires (wasted effort)
// 4. No visible feedback on elapsed time
// 5. No exponential backoff (inefficient polling)
```

**Issues:**
1. **Premature Timeout:** 30 attempts with 1s sleep = 30s, but transaction valid for 120s
2. **Wasted Polling:** Continue polling after maxTime has passed (tx can never be included)
3. **Poor UX:** No elapsed time feedback for user
4. **Network Inefficiency:** Fixed 1s interval wastes RPC calls
5. **Ledger Close Timing:** Doesn't account for Stellar's ~5s ledger close time

**Example Failure Scenario:**
```
Transaction submitted with maxTime = now + 120s
Current polling: 30 attempts × 1s = 30s total
After 30s: "Transaction pending" error
Reality: Transaction is still valid and may confirm in next 90s!
```

**Impact:**
- High: Unnecessary transaction failures
- UX: Users don't know if they should wait or retry
- Network: Inefficient RPC usage

### Technical Analysis

**Stellar Transaction Timebounds:**
```typescript
interface TimeBounds {
  minTime: string; // Unix timestamp (usually 0)
  maxTime: string; // Unix timestamp (now + 120s typical)
}

// Transaction valid if: minTime <= ledgerCloseTime <= maxTime
// Ledger closes every ~5 seconds
// After maxTime: Transaction CANNOT be included
```

**Optimal Polling Strategy:**
1. **Use maxTime as deadline:** Poll until `now > maxTime + marginSeconds`
2. **Exponential backoff:** Start fast (500ms), increase to max (5s)
3. **Ledger-aware intervals:** Align with ~5s ledger close time
4. **Elapsed time feedback:** Show user how long they've been waiting

**Exponential Backoff Formula:**
```
interval = min(initialDelay * (backoffFactor ^ attempt), maxDelay)

Example:
  Attempt 0: 500ms
  Attempt 1: 1000ms (500 * 2)
  Attempt 2: 2000ms (1000 * 2)
  Attempt 3: 4000ms (2000 * 2)
  Attempt 4+: 5000ms (capped at max)
```

#### Phase 1: Timebound-Aware Polling Function

**File:** `lib/soroban/polling.ts`
```typescript
import { SorobanRpc } from '@stellar/stellar-sdk';

export interface TimeboundPollingConfig {
  initialDelayMs?: number;      // Default: 500ms
  maxDelayMs?: number;           // Default: 5000ms (ledger close time)
  backoffFactor?: number;        // Default: 2
  ledgerCloseMarginMs?: number;  // Default: 10000ms (2 ledgers)
  onProgress?: (elapsed: number, deadline: number) => void;
}

export interface PollingResult {
  status: 'SUCCESS' | 'TIMEOUT' | 'FAILED';
  response?: SorobanRpc.Api.GetTransactionResponse;
  elapsedMs: number;
  attempts: number;
  error?: Error;
}

/**
 * Poll for transaction finality using timebound-aware exponential backoff.
 * Continues until transaction succeeds, fails, or maxTime deadline passes.
 */
export async function pollTransactionWithTimebound(
  server: SorobanRpc.Server,
  transactionHash: string,
  timeBounds: { minTime: string; maxTime: string },
  config: TimeboundPollingConfig = {}
): Promise<PollingResult> {
  const {
    initialDelayMs = 500,
    maxDelayMs = 5000,
    backoffFactor = 2,
    ledgerCloseMarginMs = 10000, // 2 ledger closes
    onProgress
  } = config;

  // Calculate polling deadline from transaction's maxTime
  const maxTimeUnix = parseInt(timeBounds.maxTime, 10);
  const deadlineMs = maxTimeUnix * 1000 + ledgerCloseMarginMs;

  const startTime = Date.now();
  let attempt = 0;
  let currentDelayMs = initialDelayMs;

  while (Date.now() < deadlineMs) {
    // Check if we're past the deadline
    const elapsed = Date.now() - startTime;
    const timeUntilDeadline = deadlineMs - Date.now();

    // Report progress to UI
    if (onProgress) {
      onProgress(elapsed, timeUntilDeadline);
    }

    // Poll transaction status
    try {
      const response = await server.getTransaction(transactionHash);

      // Transaction found and processed
      if (response.status === SorobanRpc.Api.GetTransactionStatus.SUCCESS) {
        return {
          status: 'SUCCESS',
          response,
          elapsedMs: elapsed,
          attempts: attempt + 1
        };
      }

      // Transaction failed
      if (response.status === SorobanRpc.Api.GetTransactionStatus.FAILED) {
        return {
          status: 'FAILED',
          response,
          elapsedMs: elapsed,
          attempts: attempt + 1,
          error: new Error(`Transaction failed: ${response.resultXdr}`)
        };
      }

      // Status is NOT_FOUND - transaction still pending
      // Continue polling with exponential backoff
    } catch (error) {
      console.warn(`Polling attempt ${attempt} failed:`, error);
      // Continue polling even if RPC call fails
    }

    // Calculate next delay with exponential backoff
    const nextDelayMs = Math.min(
      currentDelayMs,
      timeUntilDeadline // Don't sleep past deadline
    );

    if (nextDelayMs > 0) {
      await sleep(nextDelayMs);
    }

    // Increase delay for next iteration (exponential backoff)
    currentDelayMs = Math.min(currentDelayMs * backoffFactor, maxDelayMs);
    attempt++;
  }

  // Deadline passed - transaction timed out
  return {
    status: 'TIMEOUT',
    elapsedMs: Date.now() - startTime,
    attempts,
    error: new Error(
      `Transaction timed out after ${((deadlineMs - startTime) / 1000).toFixed(1)}s ` +
      `(maxTime: ${new Date(maxTimeUnix * 1000).toISOString()})`
    )
  };
}

function sleep(ms: number): Promise<void> {
  return new Promise(resolve => setTimeout(resolve, ms));
}

/**
 * Extract timebounds from a built Stellar transaction.
 */
export function extractTimeBounds(
  transaction: any // TransactionBuilder result
): { minTime: string; maxTime: string } | null {
  try {
    const timeBounds = transaction.timeBounds;
    if (!timeBounds) {
      return null;
    }

    return {
      minTime: timeBounds.minTime,
      maxTime: timeBounds.maxTime
    };
  } catch (error) {
    console.error('Failed to extract timebounds:', error);
    return null;
  }
}
```

#### Phase 2: Transaction Submission with Smart Polling

```typescript
import { Transaction, SorobanRpc, Keypair } from '@stellar/stellar-sdk';
import { pollTransactionWithTimebound, extractTimeBounds } from './polling';

export interface SubmitResult {
  hash: string;
  status: 'SUCCESS' | 'TIMEOUT' | 'FAILED';
  elapsedMs: number;
  attempts: number;
  response?: SorobanRpc.Api.GetTransactionResponse;
  error?: Error;
}

/**
 * Submit transaction and poll for finality with timebound-aware backoff.
 */
export async function submitAndPollTransaction(
  server: SorobanRpc.Server,
  transaction: Transaction,
  onProgress?: (elapsed: number, deadline: number) => void
): Promise<SubmitResult> {
  // Extract timebounds from transaction
  const timeBounds = extractTimeBounds(transaction);
  if (!timeBounds) {
    throw new Error('Transaction missing timebounds - cannot determine polling deadline');
  }

  console.log('Transaction timebounds:', {
    minTime: new Date(parseInt(timeBounds.minTime) * 1000).toISOString(),
    maxTime: new Date(parseInt(timeBounds.maxTime) * 1000).toISOString()
  });

  // Submit transaction
  let sendResponse: SorobanRpc.Api.SendTransactionResponse;
  try {
    sendResponse = await server.sendTransaction(transaction);
  } catch (error) {
    return {
      hash: '',
      status: 'FAILED',
      elapsedMs: 0,
      attempts: 0,
      error: new Error(`Failed to send transaction: ${error.message}`)
    };
  }

  if (sendResponse.status === 'ERROR') {
    return {
      hash: sendResponse.hash,
      status: 'FAILED',
      elapsedMs: 0,
      attempts: 0,
      error: new Error(`Send failed: ${sendResponse.errorResult}`)
    };
  }

  console.log(`Transaction submitted: ${sendResponse.hash}`);

  // Poll for finality with timebound-aware exponential backoff
  const pollResult = await pollTransactionWithTimebound(
    server,
    sendResponse.hash,
    timeBounds,
    { onProgress }
  );

  return {
    hash: sendResponse.hash,
    status: pollResult.status,
    elapsedMs: pollResult.elapsedMs,
    attempts: pollResult.attempts,
    response: pollResult.response,
    error: pollResult.error
  };
}
```

```typescript
  deadlineMs,
  status
}: TransactionProgressProps) {

    switch (status) {
      case 'success': return '✓ Transaction confirmed';
      case 'failed': return '✗ Transaction failed';
      case 'timeout': return '⏱ Transaction timed out';
      default: return '⏳ Waiting for confirmation...';
    }
  };

  return (
    <div className="p-4 border rounded-lg">
      <div className="mb-2 flex justify-between items-center">
        <span className="font-medium">{getStatusMessage()}</span>
        <span className="text-sm text-gray-600">
          {elapsedSec}s / {deadlineSec}s
        </span>
      </div>
```

## Issue #156: Automatic Recovery from txBAD_SEQ and txTOO_LATE

### Problem Statement

**Current Behavior:**
```typescript
// User signs transaction
const signedTx = await signTransaction(tx);

// Submit fails with raw error
sendTransaction(signedTx);
// Error: "The network rejected the transaction (txBadSeq)"
// Error: "The network rejected the transaction (txTooLate)"

// User sees XDR error codes, doesn't know what to do
```

**Common Failure Scenarios:**

**1. txBAD_SEQ (Stale Sequence Number):**
```
Timeline:
  T0: App loads account (sequence = 100)
  T1: User opens Freighter in another tab
  T2: User submits tx in other app (sequence = 101)
  T3: User returns to Indexer, signs tx (sequence = 100) ← STALE!
  T4: Submission fails: txBadSeq
```

**2. txTOO_LATE (Expired Timebound):**
```
Timeline:
  T0: App builds tx (maxTime = now + 120s)
  T1: User opens Freighter (prompt appears)
  T2: User connects Ledger hardware wallet
  T3: User signs on Ledger (takes 2 minutes due to review)
  T4: Freighter returns signature
  T5: Submission fails: txTooLate (maxTime passed!)
```

**Impact:**
- High: Common user errors, especially with hardware wallets
- UX: Cryptic error messages frustrate users
- Recovery: Users don't know they can retry

### Technical Analysis

**Stellar Transaction Errors:**
```typescript
enum TransactionResultCode {
  txBAD_SEQ = 'txBadSeq',       // Sequence number mismatch
  txTOO_LATE = 'txTooLate',     // maxTime has passed
  txTOO_EARLY = 'txTooEarly',   // Before minTime
  txMISSING_OPERATION = 'txMissingOperation',
  // ... other errors
}
```

**Recovery Strategy:**

**For txBAD_SEQ:**
1. Detect error from submission response
2. Reload account to get fresh sequence number
3. Rebuild transaction with new sequence
4. Request signature again (inform user why)
5. Submit rebuilt transaction

**For txTOO_LATE:**
1. Detect error from submission response
2. Rebuild transaction with fresh timebound
3. Request signature again (inform user why)
4. Submit rebuilt transaction

**Key Insight:** Both errors are **recoverable** by rebuilding the transaction and re-requesting signature. User doesn't need to start over!

### Implementation Plan

#### Phase 1: Error Detection

**File:** `lib/soroban/errors.ts`
```typescript
export enum RecoverableErrorType {
  BAD_SEQ = 'txBadSeq',
  TOO_LATE = 'txTooLate',
  TOO_EARLY = 'txTooEarly'
}

export interface RecoverableError {
  type: RecoverableErrorType;
  originalError: any;
  recoveryAction: 'rebuild' | 'wait';
  userMessage: string;
}

/**
 * Check if a transaction error is recoverable.
 */
export function isRecoverableError(error: any): RecoverableError | null {
  // Check SendTransactionResponse error
  if (error.errorResult) {
    const resultCode = error.errorResult.result().code().name;

    if (resultCode === 'txBadSeq') {
      return {
        type: RecoverableErrorType.BAD_SEQ,
        originalError: error,
        recoveryAction: 'rebuild',
        userMessage:
          'Your account sequence number changed (you may have signed a ' +
          'transaction in another app). We\'ll rebuild the transaction with ' +
          'the latest sequence number.'
      };
    }

    if (resultCode === 'txTooLate') {
      return {
        type: RecoverableErrorType.TOO_LATE,
        originalError: error,
        recoveryAction: 'rebuild',
        userMessage:
          'The transaction timebound expired (signing took longer than 2 minutes). ' +
          'We\'ll rebuild the transaction with a fresh timebound.'
      };
    }

    if (resultCode === 'txTooEarly') {
      return {
        type: RecoverableErrorType.TOO_EARLY,
        originalError: error,
        recoveryAction: 'wait',
        userMessage:
          'Transaction submitted before its minTime. Please wait a moment and try again.'
      };
    }
  }

  // Check if error message contains error codes (fallback)
  const errorMessage = error.message || error.toString();

  if (errorMessage.includes('txBadSeq') || errorMessage.includes('BAD_SEQ')) {
    return {
      type: RecoverableErrorType.BAD_SEQ,
      originalError: error,
      recoveryAction: 'rebuild',
      userMessage:
        'Sequence number mismatch. Rebuilding transaction with current sequence...'
    };
  }

  if (errorMessage.includes('txTooLate') || errorMessage.includes('TOO_LATE')) {
    return {
      type: RecoverableErrorType.TOO_LATE,
      originalError: error,
      recoveryAction: 'rebuild',
      userMessage:
        'Transaction expired. Rebuilding with fresh timebound...'
    };
  }

  return null; // Not recoverable
}
```

#### Phase 2: Transaction Rebuilder

```typescript
import {
  Transaction,
  TransactionBuilder,
  Server,
  Account,
  Operation,
  SorobanRpc
} from '@stellar/stellar-sdk';

export interface RebuildConfig {
  server: Server;
  sorobanServer: SorobanRpc.Server;
  sourcePublicKey: string;
  networkPassphrase: string;
  baseFee: string;
}

/**
 * Rebuild a transaction after txBAD_SEQ or txTOO_LATE error.
 */
export class TransactionRebuilder {
  private config: RebuildConfig;

  constructor(config: RebuildConfig) {
    this.config = config;
  }

  /**
   * Rebuild transaction with fresh account data.
   * Fetches latest sequence number and applies fresh timebound.
   */
  async rebuild(originalTransaction: Transaction): Promise<Transaction> {
    // 1. Load account to get current sequence number
    const account = await this.config.server.loadAccount(
      this.config.sourcePublicKey
    );

    console.log('Rebuilding transaction:', {
      oldSequence: originalTransaction.sequence,
      newSequence: account.sequenceNumber(),
      oldMaxTime: originalTransaction.timeBounds?.maxTime,
      newMaxTime: Date.now() / 1000 + 120
    });

    // 2. Create new transaction builder with fresh account
    const builder = new TransactionBuilder(account, {
      fee: this.config.baseFee,
      networkPassphrase: this.config.networkPassphrase
    });

    // 3. Add all operations from original transaction
    originalTransaction.operations.forEach(op => {
      builder.addOperation(op);
    });

    // 4. Set fresh timebound (120 seconds from now)
    builder.setTimeout(120);

    // 5. Add memo if present
    if (originalTransaction.memo.type !== 'none') {
      builder.addMemo(originalTransaction.memo);
    }

    // 6. Build new transaction
    const newTransaction = builder.build();

    return newTransaction;
  }

  /**
   * Rebuild specifically for txBAD_SEQ error.
   * Fetches latest sequence number.
   */
  async rebuildForBadSeq(originalTransaction: Transaction): Promise<Transaction> {
    console.log('Rebuilding for txBadSeq...');
    return this.rebuild(originalTransaction);
  }

  /**
   * Rebuild specifically for txTOO_LATE error.
   * Applies fresh timebound.
   */
  async rebuildForTooLate(originalTransaction: Transaction): Promise<Transaction> {
    console.log('Rebuilding for txTooLate...');
    return this.rebuild(originalTransaction);
  }
}
```

```typescript
import { Transaction, SorobanRpc } from '@stellar/stellar-sdk';
import { isRecoverableError, RecoverableErrorType } from './errors';
import { TransactionRebuilder, RebuildConfig } from './rebuilder';

export interface SubmitWithRecoveryConfig extends RebuildConfig {
  maxRetries?: number; // Default: 1
  onRecoveryAttempt?: (error: RecoverableError, attempt: number) => void;
  requestSignature: (tx: Transaction) => Promise<Transaction>;
}

export interface SubmitWithRecoveryResult {
  success: boolean;
  hash?: string;
  response?: SorobanRpc.Api.SendTransactionResponse;
  error?: Error;
  recoveryAttempts: number;
  recoveredFrom?: RecoverableErrorType;
}

/**
 * Submit transaction with automatic recovery from txBAD_SEQ and txTOO_LATE.
 */
export async function submitTransactionWithRecovery(
  transaction: Transaction,
  config: SubmitWithRecoveryConfig
): Promise<SubmitWithRecoveryResult> {
  const { maxRetries = 1, onRecoveryAttempt, requestSignature } = config;
  const rebuilder = new TransactionRebuilder(config);

  let currentTx = transaction;
  let recoveryAttempts = 0;
  let recoveredFrom: RecoverableErrorType | undefined;

  while (recoveryAttempts <= maxRetries) {
    try {
      // Attempt to send transaction
      const response = await config.sorobanServer.sendTransaction(currentTx);

      // Check for errors in response
      if (response.status === 'ERROR') {
        const recoverableError = isRecoverableError(response);

        if (recoverableError && recoveryAttempts < maxRetries) {
          // Error is recoverable - attempt recovery
          console.log(`Recoverable error detected: ${recoverableError.type}`);

          if (onRecoveryAttempt) {
            onRecoveryAttempt(recoverableError, recoveryAttempts + 1);
          }

          // Rebuild transaction
          const rebuiltTx = await rebuilder.rebuild(currentTx);

          // Request new signature
          console.log('Requesting signature for rebuilt transaction...');
          currentTx = await requestSignature(rebuiltTx);

          recoveryAttempts++;
          recoveredFrom = recoverableError.type;

          // Retry submission with rebuilt transaction
          continue;
        } else {
          // Error not recoverable or max retries reached
          return {
            success: false,
            error: new Error(response.errorResult?.toString() || 'Transaction failed'),
            recoveryAttempts,
            recoveredFrom
          };
        }
      }

      // Success!
      return {
        success: true,
        hash: response.hash,
        response,
        recoveryAttempts,
        recoveredFrom
      };
    } catch (error) {
      // Check if error is recoverable
      const recoverableError = isRecoverableError(error);

      if (recoverableError && recoveryAttempts < maxRetries) {
        console.log(`Recoverable error detected: ${recoverableError.type}`);

        if (onRecoveryAttempt) {
          onRecoveryAttempt(recoverableError, recoveryAttempts + 1);
        }

        // Rebuild and retry
        const rebuiltTx = await rebuilder.rebuild(currentTx);
        currentTx = await requestSignature(rebuiltTx);

        recoveryAttempts++;
        recoveredFrom = recoverableError.type;
        continue;
      } else {
        // Not recoverable or max retries reached
        return {
          success: false,
          error: error as Error,
          recoveryAttempts,
          recoveredFrom
        };
      }
    }
  }

  // Should never reach here
  return {
    success: false,
    error: new Error('Max recovery attempts exceeded'),
    recoveryAttempts,
    recoveredFrom
  };
}
```

```typescript
'use client';

import { RecoverableError, RecoverableErrorType } from '@/lib/soroban/errors';

interface TransactionRecoveryProps {
  error: RecoverableError;
  attempt: number;
  onRetry: () => void;
  onCancel: () => void;
}

export function TransactionRecovery({
  error,
  attempt,
  onRetry,
  onCancel
}: TransactionRecoveryProps) {
  const getErrorIcon = () => {
    switch (error.type) {
      case RecoverableErrorType.BAD_SEQ:
        return '🔄';
      case RecoverableErrorType.TOO_LATE:
        return '⏱️';
      default:
        return '⚠️';
    }
  };

  const getErrorTitle = () => {
    switch (error.type) {
      case RecoverableErrorType.BAD_SEQ:
        return 'Sequence Number Mismatch';
      case RecoverableErrorType.TOO_LATE:
        return 'Transaction Expired';
      case RecoverableErrorType.TOO_EARLY:
        return 'Transaction Too Early';
      default:
        return 'Transaction Error';
    }
  };

  return (
    <div className="p-4 border-2 border-yellow-400 rounded-lg bg-yellow-50">
      <div className="flex items-start gap-3">
        <span className="text-3xl">{getErrorIcon()}</span>
        <div className="flex-1">
          <h3 className="font-semibold text-lg mb-2">{getErrorTitle()}</h3>
          <p className="text-sm text-gray-700 mb-3">{error.userMessage}</p>

          {attempt > 0 && (
            <p className="text-xs text-gray-600 mb-3">
              Recovery attempt {attempt} of 1
            </p>
          )}

          <div className="flex gap-2">
            <button
              onClick={onRetry}
              className="px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700 text-sm"
            >
              {error.type === RecoverableErrorType.BAD_SEQ &&
                'Rebuild & Re-sign'}
              {error.type === RecoverableErrorType.TOO_LATE &&
                'Rebuild & Re-sign'}
              {error.type === RecoverableErrorType.TOO_EARLY && 'Wait & Retry'}
            </button>
            <button
              onClick={onCancel}
              className="px-4 py-2 bg-gray-200 text-gray-700 rounded hover:bg-gray-300 text-sm"
            >
              Cancel
            </button>
          </div>

          {error.type === RecoverableErrorType.BAD_SEQ && (
            <p className="mt-3 text-xs text-gray-600">
              💡 Tip: Avoid signing transactions in multiple apps simultaneously
              to prevent sequence conflicts.
            </p>
          )}

          {error.type === RecoverableErrorType.TOO_LATE && (
            <p className="mt-3 text-xs text-gray-600">
              💡 Tip: When using a hardware wallet (like Ledger), review and
              sign transactions within 2 minutes to avoid expiration.
            </p>
          )}
        </div>
      </div>
    </div>
  );
}
```

```typescript
import { useState } from 'react';
import { Transaction } from '@stellar/stellar-sdk';
import { submitTransactionWithRecovery } from '@/lib/soroban/submitWithRecovery';
import { RecoverableError } from '@/lib/soroban/errors';

export function useTransactionSubmit() {
  const [status, setStatus] = useState<'idle' | 'signing' | 'submitting' | 'recovering' | 'success' | 'error'>('idle');
  const [error, setError] = useState<RecoverableError | Error | null>(null);
  const [recoveryAttempt, setRecoveryAttempt] = useState(0);
  const [txHash, setTxHash] = useState<string>('');

  const submit = async (
    transaction: Transaction,
    requestSignature: (tx: Transaction) => Promise<Transaction>
  ) => {
    setStatus('signing');
    setError(null);
    setRecoveryAttempt(0);

    try {
      // Submit with automatic recovery
      const result = await submitTransactionWithRecovery(transaction, {
        server: horizonServer,
        sorobanServer: sorobanServer,
        sourcePublicKey: userPublicKey,

        baseFee: BASE_FEE,
        requestSignature,
        maxRetries: 1,
        onRecoveryAttempt: (error, attempt) => {
          setStatus('recovering');
          setError(error);
          setRecoveryAttempt(attempt);
        }
      });

      if (result.success) {
        setStatus('success');
        setTxHash(result.hash!);

        if (result.recoveredFrom) {
          console.log(`Transaction succeeded after recovering from ${result.recoveredFrom}`);
        }
      } else {
        setStatus('error');
        setError(result.error!);
      }

      return result;
    } catch (err) {
      setStatus('error');
      setError(err as Error);
      throw err;
    }
  };

  return {
    submit,
    status,
    error,
    recoveryAttempt,
    txHash
  };
}
```
